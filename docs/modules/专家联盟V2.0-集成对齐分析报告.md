#### 2.2.2 专家路由器（ExpertRouter）
**现有实现**：[expert-alliance.js](file:///d:/a10/aikjx/gitcode/infotopograph/platform/backend-node/src/expert-alliance.js) 中的 `routeExperts()` 方法

**优化路径**：
```javascript
// 现有实现：基于关键词匹配
async routeExperts(question, options) {
  const intentResult = this._detectIntent(question);
  const candidateExperts = this._matchExperts(question, intentResult);
  const scoredExperts = this._scoreExperts(candidateExperts, question);
  return scoredExperts.slice(0, options.maxExperts || 3);
}

// V2.0 优化：基于 PageRank 图谱路由
async routeExperts(question, options) {
  const queryVector = this._buildQueryVector(question);
  const pagerank = this._personalizedPageRank(
    this.capabilityGraph.adjacencyMatrix,
    queryVector,
    options.dampingFactor || 0.85
  );
  const ranked = this._rankAndCrop(pagerank, options.maxExperts);
  return ranked;
}
```

#### 2.2.3 学习闭环引擎（LearningLoopEngine）
**现有实现**：部分在 [expert-dispatcher.js](file:///d:/a10/aikjx/gitcode/infotopograph/platform/backend-node/src/expert-dispatcher.js) 中
- ✅ 已实现：性能统计、历史记录
- ❌ 缺失：轨迹压缩、Skill 提取、记忆更新

**需新增模块**：`learning-loop.js`
```javascript
class LearningLoopEngine {
  async processExecutionTrajectory(trajectory) {
    const compressed = this.compressor.compress(trajectory);
    const skills = this.extractor.extract(compressed);
    await this.memory.persistSkills(skills);
    await this.graph.updateCapabilities(skills);
    return { compressed, skills, updated: true };
  }
}
```
